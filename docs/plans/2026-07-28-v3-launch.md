# v3.0 — Show the World What We Are

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship bwb-browser v3.0 with multi-tab support, persistent authenticated sessions, one-shot `browser_act` natural language interaction, page diagnostics, realistic browser fingerprint, and a viral-ready demo — all in a clean modular architecture.

**Architecture:** Split the monolithic server.mjs into domain modules under `lib/`. Multi-tab uses CDP's target system to manage multiple page connections from a single browser instance. `browser_act` uses a rule-based DOM interaction engine (no LLM) to translate natural language instructions into CDP actions. Session persistence uses CDP cookie APIs and local JSON storage.

**Tech Stack:** Node.js 18+ ESM, chrome-remote-interface (CDP), @modelcontextprotocol/sdk, zod.

## Global Constraints

- All new files use `.mjs` extension (ESM)
- Every tool handler returns `{ content: [{ type: "text", text: JSON.stringify(result) }] }`
- Backward compatible — existing 44 tests must pass after refactoring
- `browser_act` uses heuristics/rule-based approach — NO LLM dependency in the server
- Module split: `server.mjs` stays as entry point, new code in `lib/*.mjs`
- All tools must work in single-tab mode before multi-tab is built

```js
import CDP from "chrome-remote-interface";
import { getState } from "./browser.mjs";

const tabs = new Map();   // targetId → { protocol, title, url }
let activeTabId = null;
let tabCounter = 0;

export async function ensureTabsInitialized() {
  // If no tabs, add the default one
  if (tabs.size === 0) {
    const state = getState();
    if (state.protocol) {
      // Figure out the current target's ID
      const targets = await CDP.List({ port: state.actualCdpPort || cfg.port });
      const page = targets.find(t => t.type === "page");
      if (page) {
        tabs.set(page.id, { protocol: state.protocol, title: page.title, url: page.url });
        activeTabId = page.id;
      }
    }
  }
}

export async function createTab(url) {
  const state = getState();
  const port = state.actualCdpPort || cfg.port;
  const info = await CDP.New({ port, url: url || "about:blank" });
  const protocol = await CDP({ target: info.id, port });
  tabs.set(info.id, { protocol, title: info.title || "", url: info.url || url || "" });
  activeTabId = info.id;
  return { id: info.id, title: info.title || "", url: info.url || "" };
}

export async function closeTab(targetId) {
  const state = getState();
  const port = state.actualCdpPort || cfg.port;
  if (!targetId) {
    // Close active tab
    targetId = activeTabId;
  }
  if (tabs.size <= 1) {
    throw new Error("Cannot close the only tab");
  }
  await CDP.Close({ id: targetId, port });
  const tab = tabs.get(targetId);
  if (tab && tab.protocol !== state.protocol) {
    try { await tab.protocol.close(); } catch {}
  }
  tabs.delete(targetId);
  // Switch to another tab
  if (activeTabId === targetId) {
    activeTabId = tabs.keys().next().value;
  }
  return { closed: targetId };
}

export function switchTab(targetId) {
  if (!tabs.has(targetId)) {
    throw new Error(`Tab not found: ${targetId}`);
  }
  activeTabId = targetId;
  const tab = tabs.get(targetId);
  return { id: targetId, title: tab.title, url: tab.url };
}

export function listTabs() {
  const result = [];
  for (const [id, tab] of tabs) {
    result.push({ id, title: tab.title, url: tab.url, active: id === activeTabId });
  }
  return result;
}

export function getActiveProtocol() {
  if (activeTabId && tabs.has(activeTabId)) {
    const tab = tabs.get(activeTabId);
    // The default tab uses the shared protocol
    if (tab.protocol) return tab.protocol;
  }
  return getState().protocol;
}
```

```js
import { mkdirSync, writeFileSync, readFileSync, existsSync, readdirSync } from "fs";
import { homedir } from "os";
import { join } from "path";

const SESSION_DIR = join(homedir(), ".bwb", "sessions");

function ensureDir() {
  mkdirSync(SESSION_DIR, { recursive: true });
}

export async function saveSession(name, protocol) {
  ensureDir();
  const { cookies } = await protocol.Network.getAllCookies();
  const path = join(SESSION_DIR, `${name}.json`);
  writeFileSync(path, JSON.stringify({ name, cookies, savedAt: Date.now() }, null, 2));
  return { savedTo: path, cookieCount: cookies.length };
}

export async function loadSession(name, protocol) {
  const path = join(SESSION_DIR, `${name}.json`);
  if (!existsSync(path)) {
    throw new Error(`Session not found: ${name}`);
  }
  const data = JSON.parse(readFileSync(path, "utf8"));
  // Set all cookies in the browser
  await protocol.Network.setCookies({ cookies: data.cookies });
  return { loaded: name, cookieCount: data.cookies.length, savedAt: new Date(data.savedAt).toISOString() };
}

export function listSessions() {
  ensureDir();
  const files = readdirSync(SESSION_DIR).filter(f => f.endsWith(".json"));
  return files.map(f => {
    try {
      const data = JSON.parse(readFileSync(join(SESSION_DIR, f), "utf8"));
      return { name: data.name, cookieCount: data.cookies?.length || 0, savedAt: new Date(data.savedAt).toISOString() };
    } catch {
      return { name: f.replace(".json", "") };
    }
  });
}
```

```js
export async function diagnosePage(protocol) {
  const { Runtime, Network, Page } = protocol;
  
  // Enable needed domains temporarily
  await Runtime.enable();
  await Network.enable();
  
  // Collect console errors
  const consoleErrors = [];
  const errorHandler = (params) => {
    if (params.type === "error") {
      consoleErrors.push(params.args?.map(a => a.value).join(" ") || "");
    }
  };
  Runtime.consoleAPICalled(errorHandler);
  
  // Page load metrics
  const { result: perfResult } = await Runtime.evaluate({
    expression: `JSON.stringify({
      loadTime: performance.timing ? performance.timing.loadEventEnd - performance.timing.navigationStart : null,
      domContentLoaded: performance.timing ? performance.timing.domContentLoadedEventEnd - performance.timing.navigationStart : null,
      domInteractive: performance.timing ? performance.timing.domInteractive - performance.timing.navigationStart : null,
      ttfb: performance.timing ? performance.timing.responseStart - performance.timing.requestStart : null,
    })`,
    returnByValue: true
  });
  
  // Check for broken images
  const { result: imgResult } = await Runtime.evaluate({
    expression: `JSON.stringify(Array.from(document.querySelectorAll('img')).map(i => ({
      src: i.src,
      broken: i.naturalWidth === 0,
      alt: i.alt || '(missing)'
    })).filter(i => i.broken))`,
    returnByValue: true
  });
  
  // Check meta tags
  const { result: metaResult } = await Runtime.evaluate({
    expression: `JSON.stringify({
      title: document.title,
      description: document.querySelector('meta[name=description]')?.content || '',
      viewport: document.querySelector('meta[name=viewport]')?.content || '',
      charset: document.characterSet,
    })`,
    returnByValue: true
  });
  
  // Network errors (from CDP Network events buffer)
  // ... collect failed requests
  
  return {
    url: await Runtime.evaluate({ expression: "location.href" }).then(r => r.result?.value),
    performance: JSON.parse(perfResult?.value || "{}"),
    meta: JSON.parse(metaResult?.value || "{}"),
    brokenImages: JSON.parse(imgResult?.value || "[]"),
    consoleErrors: consoleErrors.slice(0, 20),
  };
}
```

```js
// lib/act.mjs — Natural language page interaction engine
// Rule-based: no LLM dependency. Parses common instruction patterns
// and maps them to CDP DOM interaction primitives.

async function findElementByText(Runtime, text, tag = "*") {
  const { result } = await Runtime.evaluate({
    expression: `(() => {
      const items = document.querySelectorAll('${tag}');
      const lower = ${JSON.stringify(text.toLowerCase())};
      for (const el of items) {
        const t = (el.textContent || '').trim().toLowerCase();
        const aria = el.getAttribute('aria-label') || '';
        const placeholder = el.getAttribute('placeholder') || '';
        if (t.includes(lower) || aria.toLowerCase().includes(lower) || placeholder.toLowerCase().includes(lower)) {
          return { found: true, tag: el.tagName, text: (el.textContent || '').trim().slice(0, 100), selector: buildSelector(el) };
        }
      }
      return { found: false };
      function buildSelector(el) {
        if (el.id) return '#' + el.id;
        if (el.getAttribute('data-testid')) return '[data-testid="' + el.getAttribute('data-testid') + '"]';
        return el.tagName.toLowerCase() + (el.className ? '.' + (el.className || '').split(' ').filter(Boolean).join('.') : '');
      }
    })()`,
    returnByValue: true
  });
  return result?.value || { found: false };
}

async function findInput(Runtime, label) {
  const { result } = await Runtime.evaluate({
    expression: `(() => {
      const inputs = document.querySelectorAll('input, textarea, [contenteditable=true]');
      const lower = ${JSON.stringify((label || '').toLowerCase())};
      for (const el of inputs) {
        const placeholder = (el.placeholder || '').toLowerCase();
        const ariaLabel = (el.getAttribute('aria-label') || '').toLowerCase();
        const name = (el.name || '').toLowerCase();
        const id = (el.id || '').toLowerCase();
        // Check for associated label
        let labelText = '';
        if (el.id) {
          const lbl = document.querySelector('label[for="' + el.id + '"]');
          if (lbl) labelText = lbl.textContent.toLowerCase();
        }
        if (placeholder.includes(lower) || ariaLabel.includes(lower) || labelText.includes(lower) || name.includes(lower) || id.includes(lower)) {
          return { found: true, selector: buildSelector(el) };
        }
      }
      return { found: false };
      function buildSelector(el) {
        if (el.id) return '#' + el.id;
        if (el.name) return input + '[name="' + el.name + '"]';
        if (el.placeholder) return input + '[placeholder="' + el.placeholder + '"]';
        return el.tagName.toLowerCase();
      }
    })()`,
    returnByValue: true
  });
  return result?.value || { found: false };
}

export async function executeInstruction(protocol, instruction) {
  const { Runtime, Page, Input } = protocol;
  const lower = instruction.toLowerCase();
  
  // Pattern 1: "search for X" or "search X" → find search input, type, submit
  const searchMatch = lower.match(/search (?:for )?(.+)/);
  if (searchMatch) {
    const query = searchMatch[1];
    // Find search input
    let inputInfo = await findInput(Runtime, "search");
    if (!inputInfo.found) {
      inputInfo = await findInput(Runtime, "query");
    }
    if (!inputInfo.found) {
      // Try generic input
      const { result } = await Runtime.evaluate({
        expression: `document.querySelector('input[type=search], input[name=q], input[name=search], input[placeholder*=search i]') ? 
          { found: true, selector: 'input[type=search], input[name=q]' } : { found: false }`,
        returnByValue: true
      });
      inputInfo = result?.value || { found: false };
    }
    if (inputInfo.found) {
      // Type the query
      await Runtime.evaluate({
        expression: `document.querySelector('${inputInfo.selector}')?.focus()`,
      });
      await Input.insertText({ text: query });
      await new Promise(r => setTimeout(r, 300));
      // Press Enter
      await Input.dispatchKeyEvent({ type: "keyDown", key: "Enter", windowsVirtualKeyCode: 13 });
      await Input.dispatchKeyEvent({ type: "keyUp", key: "Enter", windowsVirtualKeyCode: 13 });
      await new Promise(r => setTimeout(r, 1500));
      // Get page title and content
      const { result: titleResult } = await Runtime.evaluate({ expression: "document.title" });
      const { result: bodyResult } = await Runtime.evaluate({ expression: "document.body?.textContent?.trim().slice(0, 2000) || ''" });
      return { action: "search", query, title: titleResult?.value, content: bodyResult?.value?.slice(0, 500) };
    }
  }
  
  // Pattern 2: "click the X button/link" → find by text, click
  const clickMatch = lower.match(/click (?:the |on )?(.+?)(?: button| link| element)?$/);
  if (clickMatch) {
    const target = clickMatch[1];
    // Try buttons first, then links, then generic
    let el = await findElementByText(Runtime, target, "button, [role=button], input[type=submit]");
    if (!el.found) el = await findElementByText(Runtime, target, "a");
    if (!el.found) el = await findElementByText(Runtime, target);
    if (el.found) {
      await Runtime.evaluate({
        expression: `document.querySelector(${JSON.stringify(el.selector)})?.click()`,
      });
      await new Promise(r => setTimeout(r, 1000));
      const { result: titleResult } = await Runtime.evaluate({ expression: "document.title" });
      return { action: "click", target, clicked: el.text, title: titleResult?.value };
    }
  }
  
  // Pattern 3: "go to / navigate to URL"
  const urlMatch = lower.match(/(?:go to|navigate to|open|visit) (.+)/);
  if (urlMatch) {
    let url = urlMatch[1];
    if (!url.startsWith("http")) url = "https://" + url;
    await Page.navigate({ url });
    await new Promise(r => setTimeout(r, 2000));
    const { result: titleResult } = await Runtime.evaluate({ expression: "document.title" });
    return { action: "navigate", url, title: titleResult?.value };
  }
  
  // Pattern 4: "get/extract/find X" → extract content
  const extractMatch = lower.match(/(?:get me|extract|find|what is|show me) (.+)/);
  if (extractMatch) {
    const target = extractMatch[1];
    // Try to find element containing that text
    const el = await findElementByText(Runtime, target);
    if (el.found) {
      return { action: "extract", target, content: el.text };
    }
    // Return page summary
    const { result: titleResult } = await Runtime.evaluate({ expression: "document.title" });
    const { result: bodyResult } = await Runtime.evaluate({ expression: "document.body?.textContent?.trim().slice(0, 2000) || ''" });
    return { action: "extract", target, pageTitle: titleResult?.value, pageContent: bodyResult?.value?.slice(0, 500) };
  }
  
  // Fallback: default behavior — click the most obvious interactive element
  return { action: "unknown", instruction, hint: "Try: 'search for X', 'click the Y button', 'go to URL', 'extract Z'" };
}
```
